WITH subscription_events_filtered AS 
(
    SELECT
        subscription_id,
        DATE_TRUNC('month', event_date::DATE)::DATE AS mrr_month,
        event_type,
        mrr_change
    FROM subscription_events
    WHERE event_date::DATE BETWEEN '2025-01-01' AND '2025-12-31'
),

subscription_events_aggregated_metrics AS 
(
    SELECT
        mrr_month,
        event_type,
        SUM(mrr_change) AS total_mrr
    FROM subscription_events_filtered
    GROUP BY mrr_month, event_type
),

subscription_events_pivoted AS
(
    SELECT
        mrr_month,
		
        SUM(CASE WHEN event_type = 'new' 			THEN total_mrr ELSE 0 END) AS new_mrr,
        SUM(CASE WHEN event_type = 'expansion' 		THEN total_mrr ELSE 0 END) AS expansion_mrr,
        SUM(CASE WHEN event_type = 'contraction' 	THEN total_mrr ELSE 0 END) AS contraction_mrr,
        SUM(CASE WHEN event_type = 'churn' 			THEN total_mrr ELSE 0 END) AS churn_mrr,
		
		SUM(total_mrr) AS total_mrr
		
    FROM subscription_events_aggregated_metrics
    GROUP BY mrr_month
)

SELECT * 
FROM subscription_events_pivoted
ORDER BY mrr_month;
