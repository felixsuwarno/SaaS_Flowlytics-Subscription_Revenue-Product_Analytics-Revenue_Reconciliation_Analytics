WITH subscriptions_filtered AS
(
    SELECT
        subscription_id,
        customer_id,
        plan_tier
    FROM subscriptions
    WHERE subscription_status = 'active'
      AND subscription_start <= '2025-12-31'
),
events_filtered_churned AS
(
    SELECT
        subscription_id
    FROM subscription_events
    WHERE event_type = 'churn'
      AND event_date < '2025-12-01'
),
subscriptions_churn_excluded AS
(
    SELECT
        s.subscription_id,
        s.customer_id,
        s.plan_tier
    FROM subscriptions_filtered s
    LEFT JOIN events_filtered_churned c
        ON s.subscription_id = c.subscription_id
    WHERE c.subscription_id IS NULL
),
invoices_filtered_dec2025 AS
(
    SELECT
        subscription_id
    FROM billing_invoices
    WHERE invoice_month >= '2025-12-01'
      AND invoice_month < '2026-01-01'
),
subscriptions_invoices_gap AS
(
    SELECT
        s.subscription_id,
        s.customer_id,
        s.plan_tier
    FROM subscriptions_churn_excluded s
    LEFT JOIN invoices_filtered_dec2025 i
        ON s.subscription_id = i.subscription_id
    WHERE i.subscription_id IS NULL
),
gap_expected_amount AS
(
    SELECT
        plan_tier,
        subscription_id,
        customer_id,
        CASE
            WHEN plan_tier = 'Starter'  THEN 29
            WHEN plan_tier = 'Growth'   THEN 79
            WHEN plan_tier = 'Business' THEN 199
        END AS expected_amount
    FROM subscriptions_invoices_gap
    ORDER BY plan_tier, subscription_id
)
SELECT *
FROM gap_expected_amount;
